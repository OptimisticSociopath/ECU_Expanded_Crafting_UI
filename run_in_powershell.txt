
& {
    $ErrorActionPreference = 'Stop'
    Set-StrictMode -Version Latest

    try {
        $downloads = Join-Path $env:USERPROFILE 'Downloads'
        $documents = [Environment]::GetFolderPath('MyDocuments')
        $spt = 'C:\SPT'

        $source = Join-Path $downloads 'ECU-v1.0.0-SOURCE.zip'
        $destination = Join-Path $downloads 'ECU-v1.0.0.zip'

        $lab = Join-Path $documents 'ECU_LAB'
        $runId = (Get-Date -Format 'yyyyMMdd-HHmmss') + '-' +
            [Guid]::NewGuid().ToString('N').Substring(0,8)

        $work = Join-Path $lab $runId

        Write-Host 'STARTING|ECU_1.0.0_GUID_FIX' -ForegroundColor Cyan

        if ($PSVersionTable.PSVersion.Major -ne 5 -or
            $PSVersionTable.PSVersion.Minor -ne 1 -or
            -not [Environment]::Is64BitProcess) {
            throw 'WINDOWS_POWERSHELL_5_1_X64_REQUIRED'
        }

        if (-not (Test-Path -LiteralPath $source -PathType Leaf)) {
            throw "SOURCE_ZIP_NOT_FOUND=$source"
        }

        if (Test-Path -LiteralPath $destination) {
            throw "OUTPUT_ALREADY_EXISTS=$destination"
        }

        if (-not (Test-Path -LiteralPath $spt -PathType Container)) {
            throw "SPT_NOT_FOUND=$spt"
        }

        if (-not (Get-Command dotnet -ErrorAction SilentlyContinue)) {
            throw 'DOTNET_SDK_NOT_FOUND'
        }

        Write-Host 'PASS|PREFLIGHT' -ForegroundColor Green
        Write-Host "SOURCE=$source"

        New-Item -ItemType Directory -Path $work -Force | Out-Null

        Expand-Archive `
            -LiteralPath $source `
            -DestinationPath $work

        Write-Host 'PASS|SOURCE_EXTRACTED' -ForegroundColor Green

        $required = @(
            'Build.ps1'
            'src\ECU.Client.csproj'
            'src\ExpandedCraftingUiPlugin.cs'
            'src\CraftingLayout.cs'
            'src\CenteredIngredientGrid.cs'
            'assets\icon.png'
            'NuGet.Config'
            'docs\INSTALL.txt'
        )

        foreach ($file in $required) {
            if (-not (Test-Path -LiteralPath (Join-Path $work $file) -PathType Leaf)) {
                throw "SOURCE_FILE_MISSING=$file"
            }
        }

        Write-Host 'PASS|SOURCE_STRUCTURE' -ForegroundColor Green

        $plugin = Get-Content -LiteralPath (
            Join-Path $work 'src\ExpandedCraftingUiPlugin.cs'
        ) -Raw

        if (-not $plugin.Contains(
            'public const string PluginId = "com.vargor.ecu";'
        )) {
            throw 'INCORRECT_SOURCE_GUID'
        }

        if ($plugin.Contains(
            'com.optimisticsociopath.expanded-crafting-ui'
        )) {
            throw 'OLD_GUID_STILL_PRESENT'
        }

        if (-not $plugin.Contains(
            '[BepInPlugin(PluginId, "[V_OS] ECU - Expanded Crafting UI", "1.0.0")]'
        )) {
            throw 'INCORRECT_PLUGIN_METADATA'
        }

        Write-Host 'PASS|SOURCE_GUID|com.vargor.ecu' -ForegroundColor Green
        Write-Host 'PASS|SOURCE_PLUGIN_METADATA' -ForegroundColor Green

        [xml]$project = Get-Content -LiteralPath (
            Join-Path $work 'src\ECU.Client.csproj'
        ) -Raw

        if ($project.Project.PropertyGroup.Authors -ne 'Vargor') {
            throw 'INCORRECT_AUTHOR'
        }

        if ($project.Project.PropertyGroup.Version -ne '1.0.0') {
            throw 'INCORRECT_VERSION'
        }

        Write-Host 'PASS|AUTHOR|Vargor' -ForegroundColor Green
        Write-Host 'PASS|VERSION|1.0.0' -ForegroundColor Green

        $builder = Join-Path $work 'Build.ps1'
        $powershell = Join-Path $PSHOME 'powershell.exe'

        $tokens = $null
        $parseErrors = $null

        [System.Management.Automation.Language.Parser]::ParseFile(
            $builder,
            [ref]$tokens,
            [ref]$parseErrors
        ) | Out-Null

        if (@($parseErrors).Count -gt 0) {
            $parseErrors | Out-Host
            throw 'BUILD_SCRIPT_PARSE_FAILED'
        }

        Write-Host 'PASS|BUILD_SCRIPT_AST' -ForegroundColor Green
        Write-Host 'STARTING|BUILD' -ForegroundColor Cyan

        Push-Location -LiteralPath $work

        try {
            & $powershell `
                -NoLogo `
                -NoProfile `
                -NonInteractive `
                -ExecutionPolicy Bypass `
                -File $builder `
                -SptRoot $spt

            $buildExit = $LASTEXITCODE

            if ($buildExit -ne 0) {
                throw "BUILD_FAILED_EXIT_CODE=$buildExit"
            }
        }
        finally {
            Pop-Location
        }

        Write-Host 'PASS|BUILD' -ForegroundColor Green

        $artifactRoot = Join-Path $work 'artifacts'

        $packages = @(
            Get-ChildItem `
                -LiteralPath $artifactRoot `
                -File `
                -Recurse |
                Where-Object {
                    $_.Name -in @(
                        'ECU-v1.0.0.zip'
                        'ECU-v1.0.0-SSPTMM.zip'
                    )
                }
        )

        if ($packages.Count -ne 1) {
            throw "UNEXPECTED_PACKAGE_COUNT=$($packages.Count)"
        }

        $zip = $packages[0].FullName

        Add-Type -AssemblyName System.IO.Compression.FileSystem

        $archive = [IO.Compression.ZipFile]::OpenRead($zip)

        try {
            $expected = @(
                'BepInEx/plugins/ECU/ECU.Client.dll'
                'BepInEx/plugins/ECU/icon.png'
                'ECU-INSTALL.txt'
            )

            $entries = @(
                $archive.Entries |
                    Where-Object {
                        -not [string]::IsNullOrEmpty($_.Name)
                    }
            )

            $actual = @(
                $entries |
                    ForEach-Object {
                        $_.FullName.Replace('\', '/')
                    }
            )

            if ($actual.Count -ne 3) {
                throw 'PACKAGE_ENTRY_COUNT_MISMATCH'
            }

            $differences = @(
                Compare-Object `
                    -ReferenceObject $expected `
                    -DifferenceObject $actual `
                    -CaseSensitive
            )

            if ($differences.Count -ne 0) {
                $differences | Out-Host
                throw 'PACKAGE_STRUCTURE_MISMATCH'
            }

            foreach ($entry in $entries) {
                if ($entry.Length -le 0) {
                    throw "EMPTY_PACKAGE_ENTRY=$($entry.FullName)"
                }

                $stream = $entry.Open()

                try {
                    $stream.CopyTo([IO.Stream]::Null)
                }
                finally {
                    $stream.Dispose()
                }
            }
        }
        finally {
            $archive.Dispose()
        }

        Write-Host 'PASS|PACKAGE_STRUCTURE|3/3' -ForegroundColor Green

        [IO.File]::Copy($zip, $destination, $false)

        if ((Get-Item -LiteralPath $zip).Length -ne
            (Get-Item -LiteralPath $destination).Length) {
            throw 'EXPORT_FILE_SIZE_MISMATCH'
        }

        Write-Host ''
        Write-Host 'PASS|ZIP_EXPORT' -ForegroundColor Green
        Write-Host 'PASS|NO_INSTALL_PERFORMED' -ForegroundColor Green

        Write-Host ''
        Write-Host '============================================'
        Write-Host ' MOD: [V_OS] ECU - Expanded Crafting UI'
        Write-Host ' AUTHOR: Vargor'
        Write-Host ' VERSION: 1.0.0'
        Write-Host ' GUID: com.vargor.ecu'
        Write-Host ' TYPE: Client only'
        Write-Host '============================================'
        Write-Host ''
        Write-Host "LAB=$work"
        Write-Host "READY=$destination"
        Write-Host ''
        Write-Host 'PASS_READY_MOD_ZIP' -ForegroundColor Green

        Start-Process explorer.exe -ArgumentList (
            '/select,"' + $destination + '"'
        )
    }
    catch {
        Write-Host ''
        Write-Host "FAIL_CLOSED=$($_.Exception.Message)" -ForegroundColor Red
        throw
    }
}
